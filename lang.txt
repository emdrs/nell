// Structs
struct Vector2 {
    float x = 0; // Default values
    float y;

    // Method just inject a pointer of struct as first parameter
    float magnitude() {
        return sqrt((self.x * self.x) + (self.y * self.y));
    }

    Vector2 normalized() {
        float mag = self.magnitude();
        return (Vector2) { self.x / mag, self.y / mag };
    }

    void normalize() {
        float mag = self.magnitude();
        self.x /= mag;
        self.y /= mag;
    }
}

// Possible translation
struct Vector2 {
    float x;
    float y;
}

// Method just inject a pointer of struct as first parameter
float vec2_magnitude(Vector2 *self) {
    return sqrt((self->x * self->x) + (self->y * self->y));
}

Vector2 vec2_normalized(Vector2 *self) {
    float mag = vec2_magnitude(self);
    return (Vector2) { self->x / mag, self->y / mag };
}

void vec2_normalize(Vector2 *self) {
    float mag = vec2_magnitude(self);
    self->x /= mag;
    self->y /= mag;
}

// Examples

struct Circle {
    Vector2 position;
    float radius;
};

int collide(Circle a, Circle b) {
    return (a.position - b.position).magnitude() <= (a.radius + b.radius);
}

struct Rectangle {
    Vector2 position;
    float size;
};

int collide(Rectangle a, Rectangle b) {
    float a_half_size = a.size / 2;
    float a_min_x = a.position.x - a_half_size;
    float a_max_x = a.position.x + a_half_size;
    float a_min_y = a.position.y - a_half_size;
    float a_max_y = a.position.y + a_half_size;

    float b_half_size = b.size / 2;
    float b_min_x = b.position.x - b_half_size;
    float b_max_x = b.position.x + b_half_size;
    float b_min_y = b.position.y - b_half_size;
    float b_max_y = b.position.y + b_half_size;

    return a_min_x < b_max_x &&
           a_max_x > b_min_x &&
           a_min_y < b_max_y &&
           a_max_y > b_min_y;
}

int collide(Rectangle a, Circle b) {
    float a_half_size = a.size / 2;
    float a_min_x = a.position.x - a_half_size;
    float a_max_x = a.position.x + a_half_size;
    float a_min_y = a.position.y - a_half_size;
    float a_max_y = a.position.y + a_half_size;

    float x = max(a_min_x, min(b.position.x, a_max_x));
    float y = max(a_min_y, min(b.position.y, a_max_y));
    
    float distance_x = b.position.x - x;
    float distance_y = b.position.y - y;
    
    float square_distance = (distancia_x * distance_x) + (distancia_y * distancia_y);
    
    return square_distance <= (b.radius * b.radius);
}

typedef struct signal {
    ArrayList *connections;
} Signal;

void signal_connect(Signal *signal, SignalCollidedCallback *callback)
{
    array_list_add(signal->connections, callback);
}
// End pre coded c

Signal collided = {
    array_list_create(sizeof(SignalCollidedCallback), 1)
};

typedef void (*SignalCollidedCallback)(Collision);

void signal_collided_emit(Signal *signal, Collision collision)
{
    for(int i = 0; i < signal->connections->size; i++)
        ((SignalCollidedCallback *) array_list_get(signal->connections, i))(collision);
}

// Emit
